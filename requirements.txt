# SmartCampus College Management System
# A Flask web application for Admin, Teacher, and Student portals.
# Features include authentication, attendance, marks/results, profiles,
# MySQL data storage, and Excel attendance report generation.

# Web framework
Flask

# MySQL integration for Flask
Flask-MySQLdb

# Read and generate Excel attendance workbooks
openpyxl

# Load configuration from a .env file
python-dotenv

# CSRF protection for forms and AJAX
Flask-WTF
